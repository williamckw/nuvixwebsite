# Put this inside your server { } block for nuvix.com.my
root /www/wwwroot/nuvix.com.my;
index index.html;
autoindex off;

location ~ /\.(git|env|svn) { deny all; }
location ~* \.(zip|sql|bak|log|map|md|txt|json|lock|yaml)$ { deny all; }
location ~* /(README|package|composer|vite\.config|webpack\.) { deny all; }

location / {
    try_files $uri $uri.html $uri/ =404;
}

location ~* \.(css|js|svg|png|jpg|jpeg|webp|ico)$ {
    expires 30d;
    add_header Cache-Control "public, max-age=2592000";
}

add_header X-Content-Type-Options nosniff always;
add_header X-Frame-Options SAMEORIGIN always;
add_header Referrer-Policy strict-origin-when-cross-origin always;
add_header Permissions-Policy "geolocation=(), microphone=(), camera=()" always;
